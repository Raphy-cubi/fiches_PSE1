FILE: ./Makefile
================================================================
BUILDDIR := build
OUTDIR   := output
LATEX    := pdflatex -interaction=nonstopmode -synctex=1 -output-directory=$(BUILDDIR)

.PHONY: all a3 a4 clean

all: a3 a4

$(BUILDDIR) $(OUTDIR):
	@mkdir -p $@

a3: | $(BUILDDIR) $(OUTDIR)
	$(LATEX) -jobname=bilan_A3 "\def\PaperFormat{A3}\input{main}"
	$(LATEX) -jobname=bilan_A3 "\def\PaperFormat{A3}\input{main}"
	@cp $(BUILDDIR)/bilan_A3.pdf $(OUTDIR)/bilan_A3.pdf

a4: | $(BUILDDIR) $(OUTDIR)
	$(LATEX) -jobname=bilan_A4 main.tex
	$(LATEX) -jobname=bilan_A4 main.tex
	@cp $(BUILDDIR)/bilan_A4.pdf $(OUTDIR)/bilan_A4.pdf

clean:
	rm -rf $(BUILDDIR)
	rm -f $(OUTDIR)/*.pdf
